Cet équipement autonome doit transmettre des mesures critiques (température, télémétrie, états d'alarmes) vers le système d'information de l'entreprise.

Pour garantir la robustesse industrielle, le cahier des charges impose de répondre à trois problématiques techniques fondamentales :

/* ====================================================================

   Partie 1 : Connexion STA, Diagnostic Réseau et Métriques Radio RSSI
   Compatible : Raspberry Pi Pico W & Seeed Studio XIAO ESP32-S3

#if defined(ARDUINO_ARCH_RP2040)
  #include <WiFi.h>
  const int PIN_LED = LED_BUILTIN; // LED sur puce CYW43439
#elif defined(ESP32)
  #include <WiFi.h>
  const int PIN_LED = 21;          // LED orange intégrée XIAO ESP32-S3 (Active LOW)
#endif

// Identifiants du point d'accès Wi-Fi de la salle
const char* WIFI_SSID     = "WIFI_SALLE_CIEL";
const char* WIFI_PASSWORD = "SuperSecretCIEL2026";

void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 3000); // Attente moniteur série

  pinMode(PIN_LED, OUTPUT);
#if defined(ESP32)
  digitalWrite(PIN_LED, HIGH); // Éteint pour XIAO (anode commune)
#else
  digitalWrite(PIN_LED, LOW);  // Éteint pour Pico W
#endif

  Serial.println("\n--- [BTS CIEL] DEMARRAGE DIAGNOSTIC WI-FI STA ---");

  // 1. Forcer le mode Station (désactive le mode Point d'Accès par défaut)
  WiFi.mode(WIFI_STA);
  WiFi.disconnect(true);
  delay(200);

  Serial.print("[+] Connexion en cours au SSID : ");
  Serial.println(WIFI_SSID);

  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

  // 2. Boucle de connexion avec clignotement de la LED
  uint32_t startAttempt = millis();
  while (WiFi.status() != WL_CONNECTED && (millis() - startAttempt < 15000)) {
#if defined(ESP32)
    digitalWrite(PIN_LED, !digitalRead(PIN_LED));
#else
    digitalWrite(PIN_LED, !digitalRead(PIN_LED));
#endif
    delay(250);
    Serial.print(".");
  }

  // 3. Contrôle de l'état de connexion
  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\n[OK] Connecte avec succes !");
#if defined(ESP32)
    digitalWrite(PIN_LED, LOW);  // Fixe allumée
#else
    digitalWrite(PIN_LED, HIGH); // Fixe allumée
#endif

    // Affichage exhaustif des caractéristiques réseau de couche 2 et 3
    Serial.println("==================================================");
    Serial.print("  Adresse MAC Matérielle : "); Serial.println(WiFi.macAddress());
    Serial.print("  Adresse IP attribuée   : "); Serial.println(WiFi.localIP());
    Serial.print("  Masque de sous-réseau  : "); Serial.println(WiFi.subnetMask());
    Serial.print("  Passerelle par défaut  : "); Serial.println(WiFi.gatewayIP());
    Serial.print("  Serveur DNS Primaire   : "); Serial.println(WiFi.dnsIP());
    Serial.print("  Puissance Signal (RSSI): "); Serial.print(WiFi.RSSI()); Serial.println(" dBm");
    Serial.print("  Canal Radio Wi-Fi      : "); Serial.println(WiFi.channel());
    Serial.println("==================================================");
  } else {
    Serial.println("\n[ERREUR] Echec de connexion (Timeout 15s).");
  }
}

void loop() {
  // Surveillance régulière de la qualité radio RSSI
  if (WiFi.status() == WL_CONNECTED) {
    long rssi = WiFi.RSSI();
    Serial.print("[RSSI Monitoring] Signal actuel : ");
    Serial.print(rssi);
    Serial.print(" dBm -> Qualite : ");
    if (rssi > -55) Serial.println("Excellente (Proche AP)");
    else if (rssi > -70) Serial.println("Bonne (Usage normal)");
    else if (rssi > -85) Serial.println("Faible (Risque de deconnexion)");
    else Serial.println("Critique !");
  }
  delay(5000);
}

PARTIE 2 : Comparaison de Transport L4 : Datagrammes UDP vs Session TCP (45 min)

Dans les applications IoT industrielles, le choix entre UDP (User Datagram Protocol) et TCP (Transmission Control Protocol) a un impact direct sur la réactivité, l'autonomie sur batterie et la fiabilité des alertes.

  • UDP (Mode non connecté) : Pas de poignée de main, pas de garantie de livraison, pas d'acquittement, en-tête minimaliste (8 octets). Idéal pour le streaming de télémétrie haute fréquence ou l'envoi broadcast local.
  • TCP (Mode connecté) : Établissement préalable de connexion (3-Way Handshake SYN, SYN-ACK, ACK), garantie de livraison, réémission en cas de perte, contrôle de flux, en-tête complet (20 octets minimum). Idéal pour les commandes d'actionneurs ou les transactions critiques.

2.1. Code C++ Arduino : Émission Conjointe de Télémétrie en UDP et TCP

Fichier Arduino : 02_transport_udp_vs_tcp.ino Partie 2.1 
/* ====================================================================
   BTS CIEL - Partie 2 : Comparatif Couche Transport L4 (UDP vs TCP)
   Mesure de charge utile, d'overhead et observation du Handshake
   ==================================================================== */
#include <WiFi.h>
#include <WiFiUdp.h>

const char* WIFI_SSID     = "WIFI_SALLE_CIEL";
const char* WIFI_PASSWORD = "SuperSecretCIEL2026";

// Adresse IP de votre PC de TP (remplacer par l'IP de votre machine !)
const char* PC_TARGET_IP  = "192.168.1.50";
const uint16_t UDP_PORT   = 5005;
const uint16_t TCP_PORT   = 8080;

WiFiUDP udp;
uint32_t compteurMesure = 0;

void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 2000);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(200);
    Serial.print(".");
  }
  Serial.println("\n[OK] Wi-Fi Connecte. Pret pour les tests de transport L4.");

  // Démarrage du socket UDP
  udp.begin(UDP_PORT);
}

void loop() {
  compteurMesure++;

  // Simulation d'une mesure analogique (ex: tension capteur ou température)
  float tensionCapteur = (float)random(2000, 3300) / 1000.0; // de 2.00V à 3.30V
  String payload = "{\"id\":" + String(compteurMesure) + ",\"val\":" + String(tensionCapteur, 2) + "}";

  Serial.println("\n--------------------------------------------------");
  Serial.print("[*] Transmission Trame #"); Serial.println(compteurMesure);

  // -------------------------------------------------------------
  // 1. ENVOI EN UDP (Mode Non Connecté - Datagramme Rapide)
  // -------------------------------------------------------------
  uint32_t t_udp_start = micros();
  udp.beginPacket(PC_TARGET_IP, UDP_PORT);
  udp.print(payload);
  udp.endPacket();
  uint32_t t_udp_duration = micros() - t_udp_start;

  Serial.print("  [UDP] Envoye vers "); Serial.print(PC_TARGET_IP);
  Serial.print(":"); Serial.print(UDP_PORT);
  Serial.print(" en "); Serial.print(t_udp_duration); Serial.println(" us");

  delay(500); // Décalage pour isoler les trames sous Wireshark

  // -------------------------------------------------------------
  // 2. ENVOI EN TCP (Mode Connecté - Handshake + Payload + ACK)
  // -------------------------------------------------------------
  WiFiClient tcpClient;
  uint32_t t_tcp_start = micros();
  if (tcpClient.connect(PC_TARGET_IP, TCP_PORT)) {
    tcpClient.println(payload);
    tcpClient.flush();
    tcpClient.stop(); // Fermeture de session TCP (FIN-ACK)
    uint32_t t_tcp_duration = micros() - t_tcp_start;
    Serial.print("  [TCP] Session etablie & donnee transmise en ");
    Serial.print(t_tcp_duration); Serial.println(" us");
  } else {
    Serial.println("  [TCP] Echec connexion serveur (Port 8080 non ecoute).");
  }

  delay(4500); // Période d'émission globale de 5 secondes
}

/* ====================================================================
   BTS CIEL - Partie 3.1 : Client HTTP REST Non Sécurisé
   Envoi d'un Token Bearer Secret et de données télémétrie sur le port 80
   ==================================================================== */
#include <WiFi.h>
#include <HTTPClient.h>

const char* WIFI_SSID     = "WIFI_SALLE_CIEL";
const char* WIFI_PASSWORD = "SuperSecretCIEL2026";

// URL cible de l'API de supervision (ex: serveur local du prof ou httpbin.org)
const char* SERVER_URL    = "http://httpbin.org/post";
const char* SECRET_TOKEN  = "Bearer BTS_CIEL_TOP_SECRET_TOKEN_42";

void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 2000);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(250);
    Serial.print(".");
  }
  Serial.println("\n[OK] Connecte. Envoi requetes HTTP en clair...");
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    HTTPClient http;
    http.begin(SERVER_URL);

    // Ajout des en-têtes HTTP (Headers)
    http.addHeader("Content-Type", "application/json");
    http.addHeader("Authorization", SECRET_TOKEN);

    // Corps de la requête (Payload JSON)
    String jsonPayload = "{\"station\":\"Lycée_CIEL\",\"pression_hpa\":1013.25,\"alarme\":false}";

    Serial.println("\n[HTTP] Emission POST vers httpbin.org...");
    int httpResponseCode = http.POST(jsonPayload);

    if (httpResponseCode > 0) {
      Serial.print("[HTTP] Code de Reponse : "); Serial.println(httpResponseCode);
      String response = http.getString();
      Serial.println("[HTTP] Reponse recue :");
      Serial.println(response.substring(0, 120) + "...");
    } else {
      Serial.print("[ERREUR HTTP] Code : "); Serial.println(httpResponseCode);
    }
    http.end();
  }
  delay(10000);
}

3.3. Code C++ : Sécurisation par Client HTTPS / TLS (Port 443)

Pour éliminer cette faille de sécurité, nous migrons vers WiFiClientSecure qui encapsule la requête HTTP dans une session chiffrée TLS v1.2 / v1.3 sur le port 443.

Fichier Arduino : 04_https_secure_tls_client.ino Partie 3.3 
/* ====================================================================
   BTS CIEL - Partie 3.3 : Sécurisation Cryptographique HTTPS (TLS)
   Chiffrement de bout en bout sur le port 443 avec WiFiClientSecure
   ==================================================================== */
#include <WiFi.h>
#include <WiFiClientSecure.h>
#include <HTTPClient.h>

const char* WIFI_SSID     = "WIFI_SALLE_CIEL";
const char* WIFI_PASSWORD = "SuperSecretCIEL2026";

// URL HTTPS (Remarquer le https:// et le port par défaut 443)
const char* SECURE_URL    = "https://httpbin.org/post";
const char* SECRET_TOKEN  = "Bearer BTS_CIEL_TOP_SECRET_TOKEN_42";

void setup() {
  Serial.begin(115200);
  while (!Serial && millis() < 2000);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  while (WiFi.status() != WL_CONNECTED) {
    delay(250);
    Serial.print(".");
  }
  Serial.println("\n[OK] Connecte. Envoi securise HTTPS...");
}

void loop() {
  if (WiFi.status() == WL_CONNECTED) {
    WiFiClientSecure secureClient;
    
    // Pour le TP : validation sans contrôle strict d'autorité racine CA
    // (En production industrielle, on intègre le certificat racine X.509)
    secureClient.setInsecure();

    HTTPClient https;
    Serial.println("\n[HTTPS] Etablissement session TLS vers https://httpbin.org...");

    uint32_t t0 = millis();
    if (https.begin(secureClient, SECURE_URL)) {
      https.addHeader("Content-Type", "application/json");
      https.addHeader("Authorization", SECRET_TOKEN);

      String jsonPayload = "{\"station\":\"Lycée_CIEL\",\"pression_hpa\":1013.25,\"alarme\":false}";

      int httpCode = https.POST(jsonPayload);
      uint32_t dureeSession = millis() - t0;

      if (httpCode > 0) {
        Serial.print("[HTTPS] Reponse recue en "); Serial.print(dureeSession); Serial.println(" ms");
        Serial.print("[HTTPS] Code HTTP : "); Serial.println(httpCode);
        String rep = https.getString();
        Serial.println("[HTTPS] Reponse chiffree correctement decapsulee.");
      } else {
        Serial.print("[ERREUR HTTPS] Code d'echec : "); Serial.println(httpCode);
      }
      https.end();
    }
  }
  delay(10000);
}
